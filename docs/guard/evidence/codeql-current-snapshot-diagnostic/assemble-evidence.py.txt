def retain(name: str, raw: bytes, *, compressed: bool = False) -> None:
    stored_name = name + '.gz' if compressed else name
    stored = gzip.compress(raw, mtime=0) if compressed else raw
    assert not compressed or gzip.decompress(stored) == raw
    (DEST / stored_name).write_bytes(stored)
    artifacts.append({
        'file': stored_name, 'bytes': len(stored), 'sha256': hashlib.sha256(stored).hexdigest(),
        'decoded_bytes': len(raw), 'decoded_sha256': hashlib.sha256(raw).hexdigest(),
        'encoding': 'gzip' if compressed else 'identity',
    })

def profile_nodes(module: ast.Module) -> dict[str, str]:
    entry = next(n for n in module.body if isinstance(n, ast.Assign)
                 and isinstance(n.targets[0], ast.Name) and n.targets[0].id == 'PROFILES')
    assert isinstance(entry.value, ast.Dict)
    return {str(ast.literal_eval(key)): ast.dump(value) for key, value in zip(entry.value.keys, entry.value.values)}
