def sha(data):
    return hashlib.sha256(data).hexdigest()

def original(path):
    return subprocess.check_output(['git', 'show', f'{SOURCE}:{path}'], cwd=ROOT)

class RemoveDiagnosticAdditions(ast.NodeTransformer):
    def __init__(self):
        self.removed = []

    def record(self, kind, node):
        self.removed.append({'kind': kind, 'line': node.lineno, 'ast': ast.dump(node, include_attributes=False)})

    def visit_ImportFrom(self, node):
        if node.module and node.module.endswith('runtime_hook_evidence_diagnostics'):
            self.record('new_diagnostics_import', node)
            return None
        return node

    def visit_FunctionDef(self, node):
        if node.name == '_record_failure_diagnostics':
            self.record('new_diagnostic_counter_method', node)
            return None
        if node.name == '_record_persistence_failure':
            assert [x.arg for x in node.args.kwonlyargs] == ['phase', 'code']
            self.record('diagnostic_keyword_parameters', node)
            node.args.kwonlyargs = []
            node.args.kw_defaults = []
        return self.generic_visit(node)

    def visit_AnnAssign(self, node):
        target = node.target
        name = target.id if isinstance(target, ast.Name) else target.attr if isinstance(target, ast.Attribute) else None
        if name in {'failure_code', 'failure_diagnostics', 'receipt_failure_diagnostics', '_failure_diagnostics', '_receipt_failure_diagnostics'}:
            self.record('diagnostic_local_or_counter_declaration', node)
            return None
        return self.generic_visit(node)

    def visit_Assign(self, node):
        if len(node.targets) == 1 and isinstance(node.targets[0], ast.Name) and node.targets[0].id == 'failure_code':
            assert isinstance(node.value, ast.Constant) and node.value.value == 'unacknowledged'
            self.record('fixed_ack_diagnostic_marker', node)
            return None
        return self.generic_visit(node)

    def visit_Expr(self, node):
        value = node.value
        if isinstance(value, ast.Call) and isinstance(value.func, ast.Attribute) and value.func.attr == '_record_failure_diagnostics':
            self.record('additive_counter_at_original_failure_site', node)
            return None
        return self.generic_visit(node)

    def visit_Call(self, node):
        if isinstance(node.func, ast.Attribute) and node.func.attr == '_record_persistence_failure':
            assert {x.arg for x in node.keywords} == {'phase', 'code'}
            self.record('diagnostic_call_arguments', node)
            node.keywords = []
        return self.generic_visit(node)

    def visit_ExceptHandler(self, node):
        if node.name == 'error':
            self.record('capture_previously_discarded_exception', node)
            node.name = None
        return self.generic_visit(node)

    def visit_Dict(self, node):
        pairs = []
        for key, value in zip(node.keys, node.values):
            if isinstance(key, ast.Constant) and key.value in {'failure_diagnostics', 'receipt_failure_diagnostics', 'evidence_failure_diagnostics'}:
                self.record('additive_stats_or_report_field', key)
            else:
                pairs.append((key, value))
        node.keys = [key for key, _ in pairs]
        node.values = [value for _, value in pairs]
        return self.generic_visit(node)

results = []
for path in ['src/codex_plugin_scanner/guard/daemon/runtime_hook_evidence_writer.py', 'ci/native_runtime/probe_native_default_auto.py']:
    before = original(path)
    after = (ROOT/path).read_bytes()
    old_tree = ast.parse(before)
    normalizer = RemoveDiagnosticAdditions()
    stripped = normalizer.visit(ast.parse(after))
    old_ast = ast.dump(old_tree, include_attributes=False)
    new_ast = ast.dump(stripped, include_attributes=False)
    result = {'path':path,'before_sha256':sha(before),'after_sha256':sha(after),'original_ast_sha256':sha(old_ast.encode()),'diagnostics_removed_ast_sha256':sha(new_ast.encode()),'equal_after_exact_listed_instrumentation_removed':old_ast==new_ast,'removed':normalizer.removed}
    results.append(result)
    if old_ast != new_ast:
        (OUT/'unexpected-original.ast').write_text(ast.unparse(old_tree))
        (OUT/'unexpected-normalized.ast').write_text(ast.unparse(stripped))
        raise AssertionError(path)
