"""Functional source witness; no timing, production edits or package execution."""
import copy
import hashlib
import json
import socket
import subprocess
import sys
import tempfile
from dataclasses import asdict
from pathlib import Path

root, output = map(Path, sys.argv[1:])
sys.path[:0] = [str(root / "src"), str(root)]
import codex_plugin_scanner.guard.runtime.supply_chain_package_eval as evaluator
from codex_plugin_scanner.guard.runtime.supply_chain_bundle import (
    SupplyChainBundleMalformedError,
    evaluate_cached_supply_chain_bundle,
    load_supply_chain_bundle_response,
    verify_supply_chain_bundle_response,
)
from codex_plugin_scanner.guard.runtime.supply_chain_bundle_models import SupplyChainBundlePackage
from codex_plugin_scanner.guard.store import GuardStore
from tests.test_guard_supply_chain_evaluator import WORKSPACE_ID, _artifact_for_targets, _bundle_response, _package
from tests.test_guard_supply_chain_bundle import _generate_key_pair, _sign_bundle_response


def identity():
    return {
        "commit": subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=root, text=True).strip(),
        "source_diff_sha256": hashlib.sha256(subprocess.check_output(["git", "diff", "HEAD", "--", "src"], cwd=root)).hexdigest(),
        "source_status": subprocess.check_output(["git", "status", "--porcelain", "--", "src"], cwd=root, text=True).splitlines(),
    }


initial = identity()
assert not initial["source_status"]
network_attempts = []


def reject_network(*args, **kwargs):
    network_attempts.append(True)
    raise AssertionError("Unexpected networking")


socket.create_connection = reject_network
socket.socket.connect = reject_network
socket.socket.connect_ex = reject_network
record = _package(ecosystem="npm", name="minimist", version="*", default_action="block")
invalid = []
for label, value in (("missing", None), ("null", None), ("empty", ""), ("whitespace", " ")):
    candidate = copy.deepcopy(record)
    if label == "missing":
        candidate.pop("version")
    else:
        candidate["version"] = value
    try:
        SupplyChainBundlePackage.from_dict(candidate)
    except SupplyChainBundleMalformedError as error:
        invalid.append({"case": label, "rejected": True, "error": str(error)})
    else:
        raise AssertionError("Unversioned regular package record accepted")

base = _bundle_response(packages=[_package(ecosystem="npm", name="anchor", version="1.0.0", default_action="block"), record])
cases = []
original_lookup = evaluator.evaluate_cached_supply_chain_bundle
calls = []


def observe(*args, **kwargs):
    result = original_lookup(*args, **kwargs)
    calls.append({"inputs": kwargs, "decision": asdict(result)})
    return result


evaluator.evaluate_cached_supply_chain_bundle = observe
with tempfile.TemporaryDirectory(prefix="rsp050-match-shape-") as temporary:
    for label in ("literal_star_regular_entry", "unversioned_emergency_deny"):
        response = copy.deepcopy(base)
        if label == "unversioned_emergency_deny":
            response["bundle"]["emergencyDenylist"] = [{"ecosystem": "npm", "name": "minimist", "namespace": None, "reason": "known_malware", "recommendedFixVersion": "1.2.9"}]
            key, _ = _generate_key_pair()
            response = _sign_bundle_response(response["bundle"], private_key_pem=key)
        parsed = load_supply_chain_bundle_response(response)
        verify_supply_chain_bundle_response(parsed, now=1779148800.0)
        api = asdict(evaluate_cached_supply_chain_bundle(parsed, package_name="minimist", package_version="1.2.8", ecosystem="npm", now=1779148800.0))
        workspace = Path(temporary) / label / "workspace"
        workspace.mkdir(parents=True)
        (workspace / "package-lock.json").write_text(json.dumps({"lockfileVersion": 3, "packages": {"node_modules/holder/node_modules/minimist": {"version": "1.2.8"}}}))
        store = GuardStore(workspace.parent / "guard")
        store.get_cloud_workspace_id = lambda: WORKSPACE_ID
        store.cache_supply_chain_bundle(WORKSPACE_ID, response, "2026-05-19T00:00:00Z")
        artifact = _artifact_for_targets("anchor@1.0.0", lockfile_paths=("package-lock.json",))
        calls.clear()
        result = evaluator.evaluate_package_request_artifact(artifact=artifact, store=store, workspace_dir=workspace, now="2026-05-19T00:00:00Z")
        with store._connect() as connection:
            evidence = [dict(row) for row in connection.execute("select * from guard_evidence order by evidence_id")]
        expected = 1 if label == "literal_star_regular_entry" else 2
        assert len(result.packages) == len(evidence) == expected
        assert calls[-1]["inputs"]["package_version"] == "1.2.8"
        assert api["reason"] == ("no_cached_match" if expected == 1 else "known_malware")
        assert api["emergency_deny"] == (expected == 2)
        cases.append({"case": label, "signature_verified": True, "api_concrete_version_decision": api, "full_lookup_observations": list(calls), "full_result": result.to_dict(), "persisted_evidence": evidence})
assert not network_attempts
final = identity()
assert initial == final
report = {"schema": "rsp050-match-shape-functional-source-witness-v1", "source": initial, "final_source": final, "harness_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(), "source_only": True, "timing_collected": False, "package_execution_claimed": False, "invalid_regular_record_versions": invalid, "full_evaluator_cases": cases, "assertions_passed": True}
output.write_text(json.dumps(report, indent=2) + "\n")
print(json.dumps({"source": initial["commit"], "output": str(output), "invalid_records_rejected": len(invalid), "full_evaluator_cases": len(cases), "assertions_passed": True}))
