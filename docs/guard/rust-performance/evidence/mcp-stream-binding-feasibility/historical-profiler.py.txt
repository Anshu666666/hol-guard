class Compare:
    __slots__ = ('reference', 'position', 'same', 'writes', 'maximum_chunk')

    def __init__(self, reference):
        assert type(reference) is bytes
        self.reference = reference
        self.position = 0
        self.same = True
        self.writes = 0
        self.maximum_chunk = 0

    def write(self, value):
        assert type(value) is bytes
        end = self.position + len(value)
        if end > original.MAX_BINDING_BYTES:
            raise ValueError('owned_preparation_binding_limit')
        self.same = self.reference.startswith(value, self.position, end) and self.same
        self.position = end
        self.writes += 1
        self.maximum_chunk = max(self.maximum_chunk, len(value))
        return len(value)


def compare(value, reference, topology=False):
    output = Compare(reference)
    encoder = original._ExactPickler(output, protocol=4)
    encoder.fast = True
    encoder.dump(value)
    result = output.same and output.position == len(reference)
    if topology:
        return {'exact_match': result, 'write_calls': output.writes,
                'largest_pickler_chunk_bytes': output.maximum_chunk,
                'serialized_bytes': output.position}
    return result


def measure(function):
    gc.collect()
    tracemalloc.start()
    before = tracemalloc.get_traced_memory()[0]
    result = function()
    current, peak = tracemalloc.get_traced_memory()
    tracemalloc.stop()
    assert result
    return {'traced_current_delta_bytes': current-before,
            'traced_peak_delta_bytes': peak-before}
