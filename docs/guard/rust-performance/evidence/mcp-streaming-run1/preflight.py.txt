from __future__ import annotations

import base64
import hashlib
import importlib.metadata
import json
import os
import platform
import stat
import subprocess
import sys
import tarfile
import time
import tomllib
from datetime import datetime, timezone
from pathlib import Path

ATTEMPT = Path(__file__).resolve().parent
GIT = Path('/dev/shm/hol-guard-takeover/work/hol-guard')
COMMITS = {'runtime': '5a5ad6d3873d83ac6411bf9204f704c449580c41',
           'oracle': 'd811b08f081d72c348b95f0cf9e45349fb1787e6'}
REPORT = ATTEMPT / 'preflight.json'


def now():
    return datetime.now(timezone.utc).isoformat()


def checkpoint(value):
    temporary = REPORT.with_suffix('.pending')
    temporary.write_text(json.dumps(value, indent=2) + '\n')
    temporary.replace(REPORT)


def command(*args):
    return subprocess.check_output(['git', '-C', str(GIT), *args])


def placement(path):
    status = path.stat()
    fs = os.statvfs(path)
    return {'path': str(path), 'resolved_path': str(path.resolve()),
            'device': status.st_dev, 'inode': status.st_ino,
            'free_bytes': fs.f_bfree * fs.f_frsize,
            'available_bytes': fs.f_bavail * fs.f_frsize}


def export(label, commit):
    destination = ATTEMPT / (label + '-frozen')
    destination.mkdir()
    tree = command('rev-parse', commit + '^{tree}').decode().strip()
    listing = command('ls-tree', '-r', '-z', '--full-tree', commit)
    entries = []
    for item in listing.split(b'\0'):
        if not item:
            continue
        metadata, name = item.split(b'\t', 1)
        mode, kind, oid = metadata.decode().split()
        if kind != 'blob' or mode not in {'100644', '100755'}:
            raise RuntimeError('streaming_preflight_unsupported_tree_entry')
        entries.append({'path': name.decode(), 'mode': mode, 'git_blob': oid})
    process = subprocess.Popen(['git', '-C', str(GIT), 'archive', '--format=tar', commit], stdout=subprocess.PIPE)
    assert process.stdout is not None
    with tarfile.open(fileobj=process.stdout, mode='r|') as archive:
        archive.extractall(destination, filter='data')
    process.stdout.close()
    if process.wait() != 0:
        raise RuntimeError('streaming_preflight_git_archive_failed')
    total = 0
    for entry in entries:
        path = destination / entry['path']
        info = path.lstat()
        if not stat.S_ISREG(info.st_mode):
            raise RuntimeError('streaming_preflight_export_not_regular')
        if bool(info.st_mode & 0o111) != (entry['mode'] == '100755'):
            raise RuntimeError('streaming_preflight_export_mode_changed')
        raw = path.read_bytes()
        actual = hashlib.sha1(b'blob ' + str(len(raw)).encode() + b'\0' + raw).hexdigest()
        if actual != entry['git_blob']:
            raise RuntimeError('streaming_preflight_export_blob_changed')
        entry.update(bytes=len(raw), sha256=hashlib.sha256(raw).hexdigest())
        total += len(raw)
    actual_files = sorted(str(path.relative_to(destination)) for path in destination.rglob('*') if path.is_file())
    if actual_files != sorted(entry['path'] for entry in entries):
        raise RuntimeError('streaming_preflight_export_file_set_changed')
    manifest = {'schema':'hol-guard-complete-tracked-export.v1', 'commit':commit,
                'tree':tree, 'tracked_files':len(entries), 'tracked_bytes':total,
                'all_git_blob_bytes_and_executable_modes_verified':True, 'files':entries}
    manifest_path = ATTEMPT / (label + '-export-before.json')
    manifest_path.write_text(json.dumps(manifest, indent=2) + '\n')
    return {'commit':commit, 'tree':tree, 'tracked_files':len(entries),
            'tracked_bytes':total, 'placement':placement(destination),
            'manifest':manifest_path.name, 'manifest_sha256':hashlib.sha256(manifest_path.read_bytes()).hexdigest()}


report = {'schema':'hol-guard-mcp-streaming-campaign-preflight.v1',
          'started_utc':now(), 'campaign_launched':False, 'qualification':False,
          'production_activation':False, 'exports':{}, 'stage':'start',
          'initial_placement':placement(ATTEMPT)}
if REPORT.exists():
    raise RuntimeError('streaming_preflight_refuses_to_overwrite')
checkpoint(report)
try:
    assert platform.python_implementation() == 'CPython'
    assert platform.python_version() == '3.12.14'
    for label, commit in COMMITS.items():
        report['stage'] = 'export_' + label
        checkpoint(report)
        report['exports'][label] = export(label, commit)
        checkpoint(report)
    runtime = ATTEMPT / 'runtime-frozen'
    oracle = ATTEMPT / 'oracle-frozen'
    plan_path = runtime / 'docs/guard/rust-performance/mcp-streaming-preparation-plan.json'
    plan_raw = plan_path.read_bytes()
    assert hashlib.sha256(plan_raw).hexdigest() == '1b4f85eb4233b4c39a06202689c331bafd69eca8523ad2a120ded9bafe39be59'
    plan = json.loads(plan_raw)
    for name, expected in plan['harness_sources_sha256'].items():
        assert hashlib.sha256((runtime/'scripts'/name).read_bytes()).hexdigest() == expected
    for name, expected in plan['public_oracle_reference_sources_sha256'].items():
        assert hashlib.sha256((oracle/'src/codex_plugin_scanner/guard'/name).read_bytes()).hexdigest() == expected
    report['plan_sha256'] = hashlib.sha256(plan_raw).hexdigest()
    report['harness_sources_sha256'] = plan['harness_sources_sha256']
    report['oracle_sources_sha256'] = plan['public_oracle_reference_sources_sha256']
    report['stage'] = 'runtime_dependencies'
    checkpoint(report)
    lock_raw = (runtime/'uv.lock').read_bytes()
    lock = tomllib.loads(lock_raw.decode())
    locked = {}
    normalize = lambda name: name.lower().replace('_','-').replace('.','-')
    for package in lock['package']:
        locked.setdefault(normalize(package['name']),set()).add(package['version'])
    packages = []
    checked_files = checked_bytes = unhashed_files = 0
    for distribution in sorted(importlib.metadata.distributions(), key=lambda item:item.metadata['Name'].lower()):
        name = distribution.metadata['Name']
        version = distribution.version
        record = {'name':name, 'version':version,
                  'version_in_uv_lock': version in locked.get(normalize(name),set()),
                  'metadata_sha256':{}, 'verified_record_files':0, 'verified_record_bytes':0,
                  'record_entries_without_hash':0}
        for filename in ('METADATA','WHEEL','RECORD','entry_points.txt'):
            raw = distribution.read_text(filename)
            if raw is not None:
                record['metadata_sha256'][filename] = hashlib.sha256(raw.encode()).hexdigest()
        for entry in distribution.files or []:
            if entry.hash is None:
                record['record_entries_without_hash'] += 1
                unhashed_files += 1
                continue
            data = Path(distribution.locate_file(entry)).read_bytes()
            digest = hashlib.new(entry.hash.mode, data).digest()
            if base64.urlsafe_b64encode(digest).rstrip(b'=').decode() != entry.hash.value:
                raise RuntimeError('streaming_preflight_dependency_record_mismatch')
            record['verified_record_files'] += 1
            record['verified_record_bytes'] += len(data)
            checked_files += 1
            checked_bytes += len(data)
        packages.append(record)
    dependency_report = {'schema':'hol-guard-python-runtime-identity.v1',
                         'python':sys.version, 'executable':str(Path(sys.executable).resolve()),
                         'executable_sha256':hashlib.sha256(Path(sys.executable).read_bytes()).hexdigest(),
                         'venv_config_sha256':hashlib.sha256((Path(sys.prefix)/'pyvenv.cfg').read_bytes()).hexdigest(),
                         'uv_lock_sha256':hashlib.sha256(lock_raw).hexdigest(),
                         'pyproject_sha256':hashlib.sha256((runtime/'pyproject.toml').read_bytes()).hexdigest(),
                         'distribution_count':len(packages), 'verified_record_files':checked_files,
                         'verified_record_bytes':checked_bytes, 'record_entries_without_hash':unhashed_files,
                         'unhashed_record_entries_not_claimed_verified':True, 'packages':packages}
    dependency_path = ATTEMPT/'python-dependencies-before.json'
    dependency_path.write_text(json.dumps(dependency_report, indent=2)+'\n')
    report['dependencies'] = {'path':dependency_path.name,
                              'sha256':hashlib.sha256(dependency_path.read_bytes()).hexdigest(),
                              'distribution_count':len(packages),'verified_record_files':checked_files,
                              'verified_record_bytes':checked_bytes,
                              'unlocked_installed_distributions':[{'name':p['name'],'version':p['version']} for p in packages if not p['version_in_uv_lock']]}
    fixtures = ATTEMPT/'fixtures'
    fixtures.mkdir()
    report['fixture_placement'] = placement(fixtures)
    report['runtime_placement'] = placement(runtime/'src')
    report['oracle_placement'] = placement(oracle/'src')
    report['host'] = {'system':platform.system(),'machine':platform.machine(),'kernel':platform.release(),
                      'visible_cpus':os.cpu_count(),'load_average':os.getloadavg(),
                      'cpu_quota':Path('/sys/fs/cgroup/cpu.max').read_text().strip(),
                      'memory_max':Path('/sys/fs/cgroup/memory.max').read_text().strip(),
                      'memory_current':Path('/sys/fs/cgroup/memory.current').read_text().strip()}
    report['available_tmpfs_bytes_before_launch'] = placement(ATTEMPT)['available_bytes']
    if report['available_tmpfs_bytes_before_launch'] < 512*1024*1024:
        raise RuntimeError('streaming_preflight_insufficient_coordinator_storage_reserve')
    report['stage'] = 'complete'
    report['preflight_passed'] = True
    report['finished_utc'] = now()
    checkpoint(report)
    print(json.dumps({'preflight_passed':True,'exports':{key:{field:value[field] for field in ('tracked_files','tracked_bytes')} for key,value in report['exports'].items()},'dependencies':report['dependencies'],'available_tmpfs_bytes':report['available_tmpfs_bytes_before_launch']}),flush=True)
except BaseException as error:
    report['preflight_passed'] = False
    report['failed_utc'] = now()
    report['failure'] = {'type':type(error).__name__,'code':str(error) if str(error).startswith('streaming_preflight_') else type(error).__name__}
    checkpoint(report)
    raise
