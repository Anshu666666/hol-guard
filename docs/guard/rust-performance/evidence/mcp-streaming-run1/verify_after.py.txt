ATTEMPT = Path(__file__).resolve().parent
LOCK = Path('/workspace/scratch/c911dc702e23/performance-measurement.lock')
REPORT = ATTEMPT / 'after-verification.json'


def sha256(data):
    return hashlib.sha256(data).hexdigest()


def now():
    return datetime.now(timezone.utc).isoformat()


def save_new(name, value):
    path = ATTEMPT / name
    with path.open('x') as handle:
        json.dump(value, handle, indent=2)
        handle.write('\n')
    return {'path': name, 'bytes': path.stat().st_size, 'sha256': sha256(path.read_bytes())}


def checkpoint(value):
    temporary = REPORT.with_suffix('.pending')
    temporary.write_text(json.dumps(value, indent=2) + '\n')
    temporary.replace(REPORT)


def verify_export(label, preflight):
    old_path = ATTEMPT / preflight['exports'][label]['manifest']
    old_raw = old_path.read_bytes()
    assert sha256(old_raw) == preflight['exports'][label]['manifest_sha256']
    before = json.loads(old_raw)
    root = ATTEMPT / (label + '-frozen')
    files = []
    mismatches = []
    for expected in before['files']:
        relative = expected['path']
        path = root / relative
        actual = {'path': relative}
        try:
            status = path.lstat()
            actual['regular_file'] = stat.S_ISREG(status.st_mode)
            actual['mode'] = '100755' if status.st_mode & 0o111 else '100644'
            if not actual['regular_file']:
                raise ValueError('tracked entry is no longer a regular file')
            data = path.read_bytes()
            actual.update(bytes=len(data), sha256=sha256(data),
                          git_blob=hashlib.sha1(b'blob ' + str(len(data)).encode() + b'\0' + data).hexdigest())
            if any(actual[key] != expected[key] for key in ('mode', 'bytes', 'sha256', 'git_blob')):
                mismatches.append({'expected': expected, 'actual': actual})
        except (OSError, ValueError) as error:
            actual['error'] = {'type': type(error).__name__, 'message': str(error)}
            mismatches.append({'expected': expected, 'actual': actual})
        files.append(actual)
    expected_names = {entry['path'] for entry in before['files']}
    extras = []
    for path in sorted(root.rglob('*')):
        if path.is_dir() and not path.is_symlink():
            continue
        relative = path.relative_to(root)
        if str(relative) in expected_names:
            continue
        status = path.lstat()
        extra = {'path': str(relative), 'regular_file': stat.S_ISREG(status.st_mode),
                 'generated_python_cache': '__pycache__' in relative.parts and path.suffix == '.pyc'}
        if extra['regular_file']:
            data = path.read_bytes()
            extra.update(bytes=len(data), sha256=sha256(data))
        elif path.is_symlink():
            extra['symlink_target'] = os.readlink(path)
        extras.append(extra)
    result = {'schema': 'hol-guard-complete-tracked-export-after.v1', 'commit': before['commit'],
              'tree': before['tree'], 'before_manifest_sha256': sha256(old_raw),
              'tracked_files': len(files), 'tracked_bytes': sum(item.get('bytes', 0) for item in files),
              'all_tracked_git_blob_bytes_sha256_and_executable_modes_unchanged': not mismatches,
              'mismatches': mismatches, 'files': files, 'untracked_entries': extras,
              'untracked_entries_are_only_regular_python_caches': all(item['regular_file'] and item['generated_python_cache'] for item in extras),
              'path': str(root), 'resolved_path': str(root.resolve()), 'device': root.stat().st_dev}
    descriptor = save_new(label + '-export-after.json', result)
    descriptor.update(tracked_files=result['tracked_files'], tracked_bytes=result['tracked_bytes'],
                      tracked_files_unchanged=not mismatches, untracked_entries=len(extras),
                      only_expected_untracked_python_caches=result['untracked_entries_are_only_regular_python_caches'])
    assert result['all_tracked_git_blob_bytes_sha256_and_executable_modes_unchanged'], 'tracked export changed'
    assert result['untracked_entries_are_only_regular_python_caches'], 'unexpected export entries'
    return descriptor


def verify_dependencies(preflight):
    before_raw = (ATTEMPT / preflight['dependencies']['path']).read_bytes()
    assert sha256(before_raw) == preflight['dependencies']['sha256']
    before = json.loads(before_raw)
    runtime = ATTEMPT / 'runtime-frozen'
    lock_raw = (runtime / 'uv.lock').read_bytes()
    locked = {}
    normalize = lambda name: name.lower().replace('_', '-').replace('.', '-')
    for package in tomllib.loads(lock_raw.decode())['package']:
        locked.setdefault(normalize(package['name']), set()).add(package['version'])
    packages = []
    checked_files = checked_bytes = unhashed_files = 0
    for distribution in sorted(importlib.metadata.distributions(), key=lambda item: item.metadata['Name'].lower()):
        name = distribution.metadata['Name']
        version = distribution.version
        record = {'name': name, 'version': version, 'version_in_uv_lock': version in locked.get(normalize(name), set()),
                  'metadata_sha256': {}, 'verified_record_files': 0, 'verified_record_bytes': 0, 'record_entries_without_hash': 0}
        for filename in ('METADATA', 'WHEEL', 'RECORD', 'entry_points.txt'):
            raw = distribution.read_text(filename)
            if raw is not None:
                record['metadata_sha256'][filename] = sha256(raw.encode())
        for entry in distribution.files or []:
            if entry.hash is None:
                record['record_entries_without_hash'] += 1
                unhashed_files += 1
                continue
            data = Path(distribution.locate_file(entry)).read_bytes()
            digest = hashlib.new(entry.hash.mode, data).digest()
            assert base64.urlsafe_b64encode(digest).rstrip(b'=').decode() == entry.hash.value, 'installed dependency RECORD mismatch'
            record['verified_record_files'] += 1
            record['verified_record_bytes'] += len(data)
            checked_files += 1
            checked_bytes += len(data)
        packages.append(record)
    after = {'schema': 'hol-guard-python-runtime-identity.v1', 'python': sys.version,
             'executable': str(Path(sys.executable).resolve()), 'executable_sha256': sha256(Path(sys.executable).read_bytes()),
             'venv_config_sha256': sha256((Path(sys.prefix) / 'pyvenv.cfg').read_bytes()),
             'uv_lock_sha256': sha256(lock_raw), 'pyproject_sha256': sha256((runtime / 'pyproject.toml').read_bytes()),
             'distribution_count': len(packages), 'verified_record_files': checked_files,
             'verified_record_bytes': checked_bytes, 'record_entries_without_hash': unhashed_files,
             'unhashed_record_entries_not_claimed_verified': True, 'packages': packages}
    descriptor = save_new('python-dependencies-after.json', after)
    descriptor.update(before_manifest_sha256=sha256(before_raw), identity_unchanged=after == before,
                      distribution_count=len(packages), verified_record_files=checked_files,
                      verified_record_bytes=checked_bytes)
    assert after == before, 'Python environment identity changed'
    return descriptor

        fs = os.statvfs(ATTEMPT)

except BaseException as error:
