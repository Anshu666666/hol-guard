"""Read-only verification of every reported match in the prepared commit.

Output is confined to this fresh review directory. No suppression is installed.
"""
from pathlib import Path
import ast
from collections import Counter
import hashlib
import json
import re
import subprocess
import tarfile
import zipfile

RAM = Path('/dev/shm/hol-guard-takeover')
REPO = RAM / 'work/hol-guard'
INPUT = RAM / 'gitleaks-prepared-647ff7dccbc2'
OUTPUT = RAM / 'gitleaks-prepared-647ff7dccbc2-review'
HEAD = '647ff7dccbc287401933ff807d11937227ac3348'
TREE = 'a01f031eebb5dfae6aa8f1776cf223a9d7a1a49f'
BASE = '4b89e0d2d496a85f04922b2e019a4aea15326bb9'
SOURCE = '9db62e8844c2ba2627f55b6b00e58cb5b175185d'
NEW_DOCUMENT = 'docs/guard/security/codeql-356-source-review.json'


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def git(*args):
    return subprocess.check_output(['git', *args], cwd=REPO)


def blob(commit, path):
    return git('show', commit + ':' + path)


def recursive_dicts(value):
    if isinstance(value, dict):
        yield value
        for child in value.values():
            yield from recursive_dicts(child)
    elif isinstance(value, list):
        for child in value:
            yield from recursive_dicts(child)


def main():
    initial_status = git('status', '--porcelain=v1')
    initial_worktree_ignore = (REPO / '.gitleaksignore').read_bytes()
    assert git('rev-parse', HEAD + '^{tree}').decode().strip() == TREE
    assert git('rev-parse', HEAD + '^').decode().strip() == SOURCE
    scan_inputs = {name: (INPUT / name).read_bytes()
                   for name in ('receipt.json', 'findings.json', 'scan.log', 'ignore')}
    scan = json.loads(scan_inputs['receipt.json'])
    findings = json.loads(scan_inputs['findings.json'])
    assert scan['head'] == HEAD and scan['base'] == BASE
    assert scan['findings'] == len(findings) == 109
    assert scan['unique_fingerprints'] == len({f['Fingerprint'] for f in findings}) == 68
    assert scan['exit_code'] == 1
    for name, field in [('findings.json', 'findings_sha256'), ('scan.log', 'log_sha256'), ('ignore', 'ignore_sha256')]:
        assert sha(scan_inputs[name]) == scan[field]
    assert len(scan_inputs['scan.log']) == scan['log_bytes']
    assert scan_inputs['ignore'] == blob(HEAD, '.gitleaksignore')
    assert '--log-opts=' + BASE + '..' + HEAD in scan['command']
    assert '--redact' in scan['command']
    assert scan['command'][scan['command'].index('--gitleaks-ignore-path') + 1] == str(INPUT / 'ignore')

    binary = Path('/tmp/hol-guard-gitleaks/gitleaks').read_bytes()
    archive = RAM / 'ci-audit-tools/gitleaks.tar.gz'
    assert sha(archive.read_bytes()) == 'fa0500f6b7e41d28791ebc680f5dd9899cd42b58629218a5f041efa899151a8e'
    with tarfile.open(archive) as tar:
        member = tar.getmember('gitleaks')
        assert member.isfile() and tar.extractfile(member).read() == binary
    assert subprocess.check_output(['/tmp/hol-guard-gitleaks/gitleaks', 'version'], text=True).strip() == '8.24.2'

    derived_path = 'docs/guard/security/gitleaks-hosted-evidence-review.json'
    derived_bytes = blob(HEAD, derived_path)
    derived = json.loads(derived_bytes)
    prior_bytes = (RAM / 'gitleaks-prepublication-d4e13547f.verified.json').read_bytes()
    prior = json.loads(prior_bytes)
    assert sha(prior_bytes) == derived['classification_source']['sha256']
    assert len(prior_bytes) == derived['classification_source']['bytes']
    assert len(prior['findings']) == 108 and prior['all_findings_classified'] is True
    prior_documents = {doc['path']: doc for doc in derived['source_documents']}
    documents = {path: blob(HEAD, path) for path in {f['File'] for f in findings}}
    assert set(documents) == set(prior_documents) | {NEW_DOCUMENT}
    for path, binding in prior_documents.items():
        assert sha(documents[path]) == binding['sha256']
        assert len(documents[path]) == binding['bytes']
        assert documents[path] == blob(derived['scan']['head'], path)

    api_bytes = (RAM / 'sonar-9db-issues.json').read_bytes()
    assert api_bytes == documents['docs/guard/security/sonar-rsp-9db-issues.json']
    api = json.loads(api_bytes)
    issues = {issue['key']: issue for issue in api['issues']}
    assert len(issues) == api['total'] == 42
    assert sha(api_bytes) == derived['public_identifier_verification']['public_response_sha256']
    artifact_api = json.loads((RAM / '9db-artifacts-api.json').read_bytes())['perf']['artifacts']
    proof_934_bytes = (RAM / 'gitleaks-prepublication-934507-verified.json').read_bytes()
    proof_934 = json.loads(proof_934_bytes)
    prior_launchers = {f['file_digest_sha256']: f for f in proof_934['findings']
                      if f.get('origin', {}).get('kind') == 'reconstructed_installed_console_launcher'}
    probe_path = 'ci/native_runtime/probe_installed_offline_secrets.py'
    probe = blob(SOURCE, probe_path)
    functions = {n.name: n for n in ast.parse(probe).body if isinstance(n, ast.FunctionDef)}
    assert ast.unparse(functions['_digest'].body[0]) == 'return hashlib.sha256(value).hexdigest()'
    assert ast.unparse(functions['_launcher_binding'].body[-1]) == 'return (_digest(content), binding)'
    attestor_ast_sha = sha(ast.dump(functions['_launcher_binding'], include_attributes=False).encode())

    codeql_review = json.loads(documents[NEW_DOCUMENT])
    alert = codeql_review['alert']
    before_bytes = blob(HEAD, alert['api_response_source'])
    before = json.loads(before_bytes)
    assert alert['api_response_source'] == 'docs/guard/security/codeql-pr-alerts-before.json'
    assert sha(before_bytes) == alert['api_response_sha256'] == 'b7f9c7512ee1e56128b7e58966ea4e86ea96c8d7a4659d6dcbe83bbc9e4d887b'
    assert len(before_bytes) == 59786
    correlation_bytes = blob(HEAD, alert['api_correlation_source'])
    correlation = json.loads(correlation_bytes)
    assert correlation['api_response']['sha256'] == sha(before_bytes)
    assert correlation['api_response']['bytes'] == len(before_bytes)
    matches = [x for x in recursive_dicts(before) if x.get('number') == 356]
    assert len(matches) == 1
    observed = matches[0]
    assert observed['rule']['id'] == alert['rule']
    assert observed['html_url'] == alert['url']
    assert observed['state'] == alert['state_at_recorded_api_read']
    assert observed['most_recent_instance']['commit_sha'] == alert['merge_commit']
    assert observed['most_recent_instance']['ref'] == alert['ref']
    assert observed['most_recent_instance']['location']['path'] == alert['source_location']['path']
    assert observed['most_recent_instance']['location']['start_line'] == alert['source_location']['line']

    results = []
    public_ids = []
    receipt_cache = {}
    for index, finding in enumerate(findings, 1):
        assert finding['Commit'] == HEAD and finding['RuleID'] == 'generic-api-key'
        assert finding['Secret'] == 'REDACTED' and 'REDACTED' in finding['Match']
        assert finding['StartLine'] == finding['EndLine']
        expected_fingerprint = f"{HEAD}:{finding['File']}:{finding['RuleID']}:{finding['StartLine']}"
        assert finding['Fingerprint'] == expected_fingerprint
        path = finding['File']
        document = documents[path]
        line = document.decode().splitlines()[finding['StartLine'] - 1]
        span = line.encode()[finding['StartColumn'] - 1:finding['EndColumn']].decode()
        entry = {
            'row': index, 'commit': HEAD, 'rule': finding['RuleID'],
            'path': path, 'line': finding['StartLine'],
            'start_column': finding['StartColumn'], 'end_column': finding['EndColumn'],
            'fingerprint': expected_fingerprint, 'document_sha256': sha(document),
            'exact_scanned_commit_document': True,
        }
        if path.startswith('docs/guard/security/sonar-rsp-9db-'):
            values = [value for value in issues if value in span]
            assert len(values) == 1
            value = values[0]
            issue = issues[value]
            entry.update(field='key', classification='public_sonar_issue_identifier',
                         identifier_sha256=sha(value.encode()),
                         reported_column_span_matches=True,
                         public_response_sha256=sha(api_bytes),
                         origin_rule=issue['rule'], origin_component=issue['component'])
            public_ids.append((path, value))
        else:
            fields = json.loads('{' + line.strip().rstrip(',') + '}')
            assert len(fields) == 1
            field, value = next(iter(fields.items()))
            assert isinstance(value, str) and re.fullmatch('[0-9a-f]{64}', value)
            assert value in span
            entry.update(field=field, file_digest_sha256=value, reported_column_span_matches=True)
            if path == NEW_DOCUMENT:
                assert finding['StartLine'] == 21 and field == 'api_response_sha256'
                assert value == sha(before_bytes)
                entry.update(classification='recomputed_retained_api_response_sha256',
                             digest_recomputed=True, verified_bytes=len(before_bytes),
                             origin={'kind': 'retained_codeql_pre_read_api_response',
                                     'response_source': alert['api_response_source'],
                                     'response_sha256': sha(before_bytes),
                                     'correlation_source': alert['api_correlation_source'],
                                     'correlation_sha256': sha(correlation_bytes),
                                     'alert_number': 356, 'alert_reference_verified': True,
                                     'separate_post_read_response_used': False})
            else:
                host_match = re.fullmatch(r'docs/guard/evidence/scanner-installed-9db/(linux|windows|macarm|macintel)-receipt.json', path)
                assert host_match
                host = host_match[1]
                if host not in receipt_cache:
                    zip_path = RAM / f'perf-9db-{host}.zip'
                    raw_zip = zip_path.read_bytes()
                    artifacts = [a for a in artifact_api if a['digest'] == 'sha256:' + sha(raw_zip)]
                    assert len(artifacts) == 1 and artifacts[0]['size_in_bytes'] == len(raw_zip)
                    with zipfile.ZipFile(zip_path) as z:
                        raw_receipt = z.read('aggregate/installed-offline-secrets.json')
                    receipt = json.loads(raw_receipt)
                    assert receipt['identity']['source_sha'] == SOURCE
                    expected_probe = probe.replace(b'\n', b'\r\n') if host == 'windows' else probe
                    assert receipt['probe_sha256'] == sha(expected_probe)
                    receipt_cache[host] = (raw_receipt, receipt, artifacts[0])
                raw_receipt, receipt, artifact = receipt_cache[host]
                assert raw_receipt == document
                origin = {'source_commit': SOURCE, 'probe_member': probe_path,
                          'probe_sha256': receipt['probe_sha256'],
                          'artifact_id': artifact['id'], 'artifact_sha256': artifact['digest'].split(':')[1],
                          'receipt_sha256': sha(raw_receipt), 'original_receipt_bytes_unchanged': True}
                if field.startswith('codex_plugin_scanner.'):
                    source_path = 'src/' + field.replace('.', '/') + '.py'
                    source_bytes = blob(SOURCE, source_path)
                    if host == 'windows':
                        assert b'\r\n' not in source_bytes
                        source_bytes = source_bytes.replace(b'\n', b'\r\n')
                    assert sha(source_bytes) == value
                    assert receipt['identity']['module_sha256'][field] == value
                    entry.update(classification='recomputed_module_sha256', digest_recomputed=True,
                                 verified_bytes=len(source_bytes))
                    origin.update(member=source_path, newline_encoding='CRLF' if host == 'windows' else 'LF')
                else:
                    assert field == 'hol-guard-secrets'
                    assert receipt['identity']['launcher_sha256'][field] == value
                    binding = receipt['identity']['launcher_binding'][field]
                    assert binding['record_verified'] is True
                    assert binding['interpreter_matches_current_installation'] is True
                    if host != 'windows':
                        reconstructed = (RAM / f'abf-scanner-{host}-launcher').read_bytes()
                        assert sha(reconstructed) == value
                        old = prior_launchers[value]
                        assert sha(reconstructed.split(b'\n', 1)[1]) == old['origin']['wrapper_body_sha256']
                        entry.update(classification='reconstructed_unix_launcher_sha256',
                                     digest_recomputed=True, verified_bytes=len(reconstructed))
                        origin.update(original_proof_sha256=sha(proof_934_bytes),
                                      original_proof_origin=old['origin'],
                                      original_launcher_bytes_downloaded=False)
                    else:
                        assert binding['uv_pe_resources'] is True
                        assert binding['windows_embedded_wrapper'] is True
                        assert binding['wrapper'] == 'uv'
                        entry.update(classification='producer_verified_windows_launcher_sha256',
                                     digest_recomputed=False, digest_producer_verified=True)
                        origin.update(producer_ast_sha256=attestor_ast_sha,
                                      producer='_launcher_binding_then_digest',
                                      installed_RECORD_and_fixed_wrapper_and_interpreter_verified=True,
                                      original_launcher_binary_retained=False)
                entry['origin'] = origin
        if path != NEW_DOCUMENT:
            prior_matches = [p for p in prior['findings'] if p['path'] == path
                             and p['line'] == finding['StartLine'] and p['field'] == entry['field']
                             and p.get('identifier_sha256', p.get('file_digest_sha256')) == entry.get('identifier_sha256', entry.get('file_digest_sha256'))]
            assert len(prior_matches) == 1
            assert prior_matches[0]['finding_document_sha256'] == sha(document)
            entry['same_prior_reviewed_bytes_independently_reverified'] = True
        results.append(entry)
    counts = dict(Counter(row['classification'] for row in results))
    assert counts == {'public_sonar_issue_identifier': 84, 'recomputed_module_sha256': 20,
                      'reconstructed_unix_launcher_sha256': 3,
                      'producer_verified_windows_launcher_sha256': 1,
                      'recomputed_retained_api_response_sha256': 1}
    assert len(results) == 109
    for path in {p for p, _ in public_ids}:
        values = [value for p, value in public_ids if p == path]
        assert len(values) == 42 and set(values) == set(issues)
    fingerprints = ('\n'.join(sorted({r['fingerprint'] for r in results})) + '\n').encode()
    assert len(fingerprints.splitlines()) == 68
    assert git('status', '--porcelain=v1') == initial_status
    assert (REPO / '.gitleaksignore').read_bytes() == initial_worktree_ignore
    proof = {
        'schema': 'hol-guard.prepared-commit-finding-review.v1',
        'commit': HEAD, 'tree': TREE, 'scan_base': BASE, 'parent': SOURCE,
        'original_scan_exit_code': scan['exit_code'],
        'original_finding_count': len(findings), 'unique_fingerprint_count': 68,
        'classification_counts': counts, 'all_109_rows_verified': True,
        'file_digest_values_recomputed': 24,
        'digest_producers_verified_without_full_binary_recomputation': 1,
        'all_digest_bytes_recomputed': False,
        'scanner': {'version': '8.24.2', 'binary_sha256': sha(binary),
                    'official_archive_sha256': sha(archive.read_bytes())},
        'scan_inputs': {name: {'bytes': len(raw), 'sha256': sha(raw)} for name, raw in scan_inputs.items()},
        'inherited_ignore_equals_prepared_commit': True,
        'previous_review': {'source': derived_path, 'source_sha256': sha(derived_bytes),
                            'detailed_proof_sha256': sha(prior_bytes),
                            'unchanged_source_documents_verified': len(prior_documents)},
        'documents': [{'path': path, 'bytes': len(raw), 'sha256': sha(raw)}
                      for path, raw in sorted(documents.items())],
        'fingerprint_list': {'path': 'verified-fingerprints.txt', 'count': 68,
                             'bytes': len(fingerprints), 'sha256': sha(fingerprints)},
        'source_or_ignore_modified': False, 'git_refs_moved': False,
        'security_alert_state_transition_performed': False,
        'security_alert_disposition_inferred': False,
        'review_scope': 'Only Gitleaks generic-api-key metadata classifications at this exact prepared commit.',
        'limitations': [
            'The original Windows launcher PE binary is absent; its immutable source, original hosted receipt, RECORD, wrapper and interpreter bindings verify the SHA256 producer without recomputing the full binary.',
            'The new CodeQL digest refers to the retained pre-read response, not the distinct later post-read response.',
            'This is an exact finding review and fingerprint list; no ignore is installed and no subsequent zero-finding scan is claimed.',
            'No determination or transition of the underlying CodeQL/Sonar security findings is part of this review.'
        ],
        'findings': results,
    }
    OUTPUT.mkdir(exist_ok=True)
    (OUTPUT / 'verified-fingerprints.txt').write_bytes(fingerprints)
    (OUTPUT / 'review.json').write_text(json.dumps(proof, sort_keys=True, indent=2) + '\n')
    for name in ('receipt.json', 'findings.json', 'scan.log'):
        (OUTPUT / ('original-' + name)).write_bytes(scan_inputs[name])
    print(json.dumps({'commit': HEAD, 'tree': TREE, 'rows_verified': len(results),
                      'fingerprints': 68, 'classifications': counts,
                      'review_sha256': sha((OUTPUT / 'review.json').read_bytes()),
                      'fingerprints_sha256': sha(fingerprints),
                      'source_or_ignore_modified': False}, sort_keys=True))


if __name__ == '__main__':
    main()
